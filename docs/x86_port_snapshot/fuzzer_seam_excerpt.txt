    def _cross_input_priming_check(self) -> None:
        """
        Perform a cross-input priming check, a generalization of `_priming_check` that replaces it
        when CONF.enable_cross_input_priming is set. The standard check keeps a violation only if
        the divergence follows the detecting input's own data; the cross-input check instead asks
        *which* input in the sequence causes it, and accepts the violation if the answer is any of
        them. See `rvzr/cross_input_priming.py` and `rvzr/cross_input.py`.
        """
        check = CrossInputPrimingCheck(self.fuzzer.executor, self.fuzzer.analyser, self.fuzzer.log)
        self.violations = check.run(self.violations, self.org_inputs, self.boosted_inputs,
                                    self.ctraces)
